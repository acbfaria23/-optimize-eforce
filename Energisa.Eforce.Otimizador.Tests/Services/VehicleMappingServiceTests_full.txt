    // ?? Ajuste de turno (�5.1) ??????????????????????????????????????????????

    [Fact]
    public void MapearVehicles_CargaTrabalho80_FimTurnoReduzido()
    {
        var equipe = TestDataFactory.CriarEquipeValida();
        equipe.PorcentagemCargaTrabalho = 80;
        equipe.InicioTurno = new DateTime(2025, 1, 1, 8, 0, 0, DateTimeKind.Utc);
        equipe.FimTurno = new DateTime(2025, 1, 1, 17, 0, 0, DateTimeKind.Utc);
        var regiao = TestDataFactory.CriarRegiao();

        var (_, equipesProcessadas) = _service.MapearVehicles([equipe], [regiao], TestDataFactory.DataExecucao);

        // 9 horas * 80% = 7.2 horas ? 08:00 + 7.2h = 15:12
        var esperado = new DateTime(2025, 1, 15, 8, 0, 0, DateTimeKind.Utc).AddHours(9 * 0.80);
        Assert.Equal(esperado, equipesProcessadas[0].FimTurnoAjustado);
    }

    [Fact]
    public void MapearVehicles_CargaTrabalho100_FimTurnoIgual()
    {
        var equipe = TestDataFactory.CriarEquipeValida();
        equipe.PorcentagemCargaTrabalho = 100;
        equipe.InicioTurno = new DateTime(2025, 1, 1, 8, 0, 0, DateTimeKind.Utc);
        equipe.FimTurno = new DateTime(2025, 1, 1, 17, 0, 0, DateTimeKind.Utc);
        var regiao = TestDataFactory.CriarRegiao();

        var (_, equipesProcessadas) = _service.MapearVehicles([equipe], [regiao], TestDataFactory.DataExecucao);

        var esperado = new DateTime(2025, 1, 15, 17, 0, 0, DateTimeKind.Utc);
        Assert.Equal(esperado, equipesProcessadas[0].FimTurnoAjustado);

    [Fact]
    public void MapearVehicles_InicioTurno_AjustadoParaDiaExecucao()
    {
        var equipe = TestDataFactory.CriarEquipeValida();
        equipe.InicioTurno = new DateTime(2020, 6, 1, 8, 0, 0, DateTimeKind.Utc);

        var (_, equipesProcessadas) = _service.MapearVehicles([equipe], [regiao], TestDataFactory.DataExecucao);
